```python
"""Item 068 whole-output comparison (scratch, uncommitted; 04 Tests 6b). stdlib only.
usage: python3 -I compare_068.py <base_dir> <head_dir>"""
import csv
import filecmp
import json
import os
import re
import sys

GAP_COUNT = re.compile(r"^earnings: (\d+) gap\(s\) > 120 days between dates$")
LISTED = "earnings: gap(s) > 120 days between listed dates ("


def fix(s):
    parts = s.split("; ")
    n = None
    keep = []
    for p in parts:
        m = GAP_COUNT.match(p)
        if m and n is None:
            n = m.group(1)
            continue
        keep.append(p)
    if n is None:
        return s
    out = []
    done = False
    for p in keep:
        if not done and p.startswith(LISTED):
            p = f"earnings: {n} " + p[len("earnings: "):]
            done = True
        out.append(p)
    if not done:
        raise SystemExit(f"FAIL: count line without a listed-dates line: {s!r}")
    return "; ".join(out)


def main(base, head):
    fails = []
    fb, fh = set(os.listdir(base)), set(os.listdir(head))
    for f in sorted(fb ^ fh):
        fails.append(f"file on one side only: {f} ({'base' if f in fb else 'head'})")
    special = {"summary.json", "bot_rule_by_ticker.csv"}
    same = 0
    for f in sorted((fb & fh) - special):
        if filecmp.cmp(os.path.join(base, f), os.path.join(head, f), shallow=False):
            same += 1
        else:
            fails.append(f"byte difference: {f}")
    print(f"files: base {len(fb)}, head {len(fh)}; byte-identical (excl. 2 compared below): {same} of {len(fb & fh - special)}")

    # summary.json
    jb = json.load(open(os.path.join(base, "summary.json")))
    jh = json.load(open(os.path.join(head, "summary.json")))
    stats = {"leaves": 0, "warn_changed": 0}
    changed_tk = []

    def walk(b, h, path):
        if isinstance(b, dict) or isinstance(h, dict):
            if not (isinstance(b, dict) and isinstance(h, dict)):
                fails.append(f"type differs at {path}")
                return
            if set(b) != set(h):
                fails.append(f"key sets differ at {path}: {sorted(set(b) ^ set(h))}")
                return
            for k in b:
                walk(b[k], h[k], path + [k])
            return
        if isinstance(b, list) or isinstance(h, list):
            if not (isinstance(b, list) and isinstance(h, list)) or len(b) != len(h):
                fails.append(f"list differs in type/length at {path}")
                return
            for i, (x, y) in enumerate(zip(b, h)):
                walk(x, y, path + [i])
            return
        stats["leaves"] += 1
        if len(path) == 3 and path[0] == "counts_bot_rule_bid" and path[2] == "warnings":
            want = fix(b)
            if want != b:
                stats["warn_changed"] += 1
                changed_tk.append(path[1])
            if repr(h) != repr(want):
                fails.append(f"warnings at {path}: head {h!r} != fix(base) {want!r}")
        elif repr(h) != repr(b):
            fails.append(f"leaf differs at {'.'.join(map(str, path))}: {b!r} -> {h!r}")

    walk(jb, jh, [])
    print(f"summary.json: leaves compared {stats['leaves']}; warnings leaves changed by fix: "
          f"{stats['warn_changed']} {sorted(changed_tk)}")

    # bot_rule_by_ticker.csv
    rb = list(csv.reader(open(os.path.join(base, "bot_rule_by_ticker.csv"), newline="")))
    rh = list(csv.reader(open(os.path.join(head, "bot_rule_by_ticker.csv"), newline="")))
    if rb[0] != rh[0]:
        fails.append("bot_rule_by_ticker.csv: header differs")
    if len(rb) != len(rh):
        fails.append(f"bot_rule_by_ticker.csv: rows {len(rb)} vs {len(rh)}")
    wi = rb[0].index("warnings")
    rows_changed = 0
    for r, (x, y) in enumerate(zip(rb[1:], rh[1:]), 1):
        if len(x) != len(y):
            fails.append(f"csv row {r}: cell count differs")
            continue
        for c, (a, b) in enumerate(zip(x, y)):
            if c == wi:
                want = fix(a)
                if want != a:
                    rows_changed += 1
                if b != want:
                    fails.append(f"csv row {r} warnings: {b!r} != fix(base) {want!r}")
            elif a != b:
                fails.append(f"csv row {r} col {rb[0][c]}: {a!r} -> {b!r}")
    print(f"bot_rule_by_ticker.csv: rows compared {len(rb) - 1}; rows changed by fix: {rows_changed}")

    for f in fails:
        print("FAIL:", f)
    print("RESULT:", "PASS" if not fails else f"FAIL ({len(fails)})")
    return 1 if fails else 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1], sys.argv[2]))
```
